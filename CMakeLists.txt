cmake_minimum_required(VERSION 3.21)
project(TinyX86 VERSION 1.0.0 LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "TinyX86 stores host addresses in uint32_t; use Windows x86 (-A Win32).")
endif()
set(CORE_X86_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(CORE_X86_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PETOOLS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(deps/Core-x86-Disassembler EXCLUDE_FROM_ALL)
add_subdirectory(deps/PETools EXCLUDE_FROM_ALL)
add_library(tinyx86_cpu STATIC TinyX86.c)
target_include_directories(tinyx86_cpu PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tinyx86_cpu PUBLIC core_x86)
if(MSVC)
  target_compile_options(tinyx86_cpu PUBLIC /utf-8)
endif()
add_executable(TinyX86 main.c debugger.c PELoader.c)
target_link_libraries(TinyX86 PRIVATE tinyx86_cpu petools psapi)
option(TINYX86_BUILD_TESTS "Build tests against the real shared decoder" ON)
if(TINYX86_BUILD_TESTS)
  enable_testing()
  add_executable(cpu_integration_tests tests/cpu_integration_tests.c)
  target_link_libraries(cpu_integration_tests PRIVATE tinyx86_cpu)
  add_test(NAME cpu_shared_decoder COMMAND cpu_integration_tests)
endif()
